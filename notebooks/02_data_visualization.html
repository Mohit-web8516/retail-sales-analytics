def discount_band(d):
    if d == 0:
        return 'No Discount'
    elif d <= 0.2:
        return 'Low (0-20%)'
    elif d <= 0.4:
        return 'Medium (20-40%)'
    else:
        return 'High (40%+)'

if 'DiscountBand' not in df.columns:
    df['DiscountBand'] = df['Discount'].apply(discount_band)

top_products = df.groupby('ProductName').agg(
